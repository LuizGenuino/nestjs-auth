16 - cria o serviço de login com jwtService e prismaService
    16.1 - abre o arquivo auth.service.ts
        import { Injectable } from '@nestjs/common';
        import { LoginDto } from './login.dto.js';
        import { JwtService } from '@nestjs/jwt';
        import { PrismaService } from '../prisma/prisma.service.js';
        import * as bcrypt from 'bcrypt'

        @Injectable()
        export class AuthService {

            constructor(private jwtService: JwtService, private prismaService: PrismaService) { }

            async login(loginDto: LoginDto) {
                const user = await this.prismaService.user.findUnique({
                    where: { email: loginDto.email }
                })

                if (!user) {
                    throw new Error("Invalid Credentials")
                }

                const isPasswordValid = bcrypt.compareSync(loginDto.password, user.password)

                if (!isPasswordValid) {
                    throw new Error("Invalid Credentials")
                }

                const token = this.jwtService.sign({
                    sub: user.id, role: user.role, name: user.name, email: user.email
                })

                return { access_token: token }
            }
        }

18 - vamos adicionar um guardião para interceptar algumas requisições para validar o acesso
    18.1 - nest g guard auth
    18.2 - acessa o arquivo auth.guard.ts
    import { CanActivate, ExecutionContext, Injectable, UnauthorizedException } from '@nestjs/common';
    import { JwtService } from '@nestjs/jwt';
    import { Request } from 'express';
    import { PrismaService } from '../prisma/prisma.service.js';

    @Injectable()
    export class AuthGuard implements CanActivate {

        constructor(private jwtService: JwtService, private prismaService: PrismaService) { }

        async canActivate(
            context: ExecutionContext,
        ): Promise<boolean> {

            const request: Request = context.switchToHttp().getRequest();
            const token = request.headers['authorization']?.split(" ")[1] // o ? valida se o dado existe antes prosseguir; split(" ") separa a string em um array em cada espaço; [1] pego a segunda posição do array

            if (!token) {
                throw new UnauthorizedException('No token provided')
            }

            try {
                const payload = this.jwtService.verify<{
                    sub: string, role: string, name: string, email: string
                }>(token, { algorithms: ['HS256'] })

                const user = await this.prismaService.user.findUnique({
                    where: { id: payload.sub }
                })

                if(!user){
                    throw new UnauthorizedException("User not found")
                }

                request.user = user

                return true
            } catch (e) {
                console.error(e)
                throw new UnauthorizedException('Invalid token', { cause: e })
            }

            return true;
        }
    }

24 - configura o casl utilizando o modelo de autorização ABAC
    24.1 - adiciona o codigo em casl-ability.service.ts
        import { Injectable, Scope } from '@nestjs/common';
        import { Post, Role, User } from '../../generated/prisma/client.js';
        import { Ability, AbilityBuilder } from '@casl/ability';
        import { createPrismaAbility, PrismaQuery, Subjects } from '@casl/prisma';

        export type PermActions = 'manage' | 'create' | 'read' | 'update' | 'delete';

        export type PermissionResource =
            | Subjects<{
                User: User;
                Post: Post;
            }>
            | 'all';

        // Na v7, usa-se Ability no lugar de PureAbility
        export type AppAbility = Ability<
            [PermActions, PermissionResource],
            PrismaQuery
        >;

        export type DefinePermissions = (
            user: User,
            builder: AbilityBuilder<AppAbility>,
        ) => void;

        const rolePermissionsMap: Record<Role, DefinePermissions> = {
            ADMIN(user, { can }) {
                can('manage', 'all');
            },
            EDITOR(user, { can }) {
                can('create', 'Post');
                can('read', 'Post');
                can('update', 'Post');
            },
            WRITER(user, { can }) {
                can('create', 'Post');
                can('read', 'Post', { authorId: user.id });
                can('update', 'Post', { authorId: user.id });
            },
            READER(user, { can }) {
                can('read', 'Post', { published: true });
            },
        };


        // modelo de autorização
        // rbac - role based access control
        // abac - attribute based access control


        @Injectable({ scope: Scope.REQUEST }) // Esse serviço vai ser gerado uma instancia por requisição
        export class CaslAbilityService {

            ability: AppAbility

            createForUser(user: User) {

                const builder = new AbilityBuilder<AppAbility>(createPrismaAbility)

                rolePermissionsMap[user.role](user, builder)

                this.ability = builder.build()

                return this.ability
            }
}

            import { ForbiddenException, Injectable } from '@nestjs/common';
            import { CreateUserDto } from './dto/create-user.dto.js';
            import { UpdateUserDto } from './dto/update-user.dto.js';
            import { PrismaService } from '../prisma/prisma.service.js';
            import * as bcrypt from 'bcrypt'
            import { CaslAbilityService } from '../casl/casl-ability/casl-ability.service.js';

            @Injectable()
            export class UsersService {

                constructor(
                    private prismaService: PrismaService,
                    private abilityService: CaslAbilityService // importa a instancia criada para o usuario no auth.guard.ts
                ) { }

                create(createUserDto: CreateUserDto) {
                    return this.prismaService.user.create({
                        data: {
                            name: createUserDto.name,
                            email: createUserDto.email,
                            password: bcrypt.hashSync(createUserDto.password, 10),
                            role: createUserDto.role,
                        },
                    });
                }

                findAll() {
                    const ability = this.abilityService.ability

                    if (!ability.can('read', 'User')) {
                        throw new ForbiddenException('Anauthorized')
                    }
                    return this.prismaService.user.findMany();
                }

                findOne(id: string) {
                    const ability = this.abilityService.ability

                    if (!ability.can('read', 'User')) {
                        throw new ForbiddenException('Anauthorized')
                    }

                    return this.prismaService.user.findUnique({
                        where: { id },
                    });
                }

                update(id: string, updateUserDto: UpdateUserDto) {
                    const ability = this.abilityService.ability

                    if (!ability.can('update', 'User')) {
                        throw new ForbiddenException('Anauthorized')
                    }

                    return this.prismaService.user.update({
                        where: { id },
                        data: updateUserDto,
                    });
                }

                remove(id: string) {
                    const ability = this.abilityService.ability

                    if (!ability.can('delete', 'User')) {
                        throw new ForbiddenException('Anauthorized')
                    }

                    return this.prismaService.user.delete({
                        where: { id },
                    });
                }
            }
