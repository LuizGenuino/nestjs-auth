nest generate (ou apenas g) resource [name ou vazio] - cria um modulo completo com dto, entidade, controller, module e service e ja adiciona no main.ts para funcioar a rota automaticamente

mas posso usar:

nest nest generate (ou apenas g) controller || module || service [name ou vazio] - para criar apenas o arquivo determinado ja integrando 

obs: caso a pasta exista cria dentro da pasta senão cria a pasta e o arquivo
ex: nest g controller users -> src/users/users.controller.ts


--- conectando com o banco de dados ----
1 - criar o container docker para rodar o banco de dados
    1.1 criar o arquivo docker-compose.yaml
    1.2 add o codigo de criação
        services:
            db:
                image: postgres:latest
                environment:
                POSTGRES_USER: postgres
                POSTGRES_PASSWORD: postgres
                POSTGRES_DB: nest_auth_db
                ports:
                - "5432:5432"
    1.3 rodar o comando docker-compose up

2 - instalar o prisma seguindo a documentação do nest e iniciar 
    2.1 - comando: npm install prisma@7 dotenv --save-dev    
    2.2 - comando: npx prisma init --datasource-provider postgresql --output ../src/generated/prisma   
    2.3 - comando: npx @prisma/client  

3 - criar o model das tabela do banco no arquivo prisma/schema.prisma
    enum Role {
    ADMIN
    EDITOR
    WRITER
    READER
    }

    model User {
    id        String   @id @default(cuid())
    name      String
    email     String   @unique
    password  String
    role      Role     @default(READER)
    createdAt DateTime @default(now())
    updatedAt DateTime @updatedAt
    }

4 - criar a migração das tabelas criadas
 4.1 comando: npx prisma migrate dev

5 - comando: npx prisma generate


6 - criar o modulo e o service do prisma para integrar ao nest
    6.1 - comando: nest g module prisma
    6.2 - comando: nest g service prisma   
    6.3 - comando npm install @nestjs/config
    6.4 adiciona no app.module.ts o trecho 
         imports: [ ConfigModule.forRoot({ isGlobal: true }),UsersModule, PrismaModule],

7 - iniciar a conexao da api com o banco no arquivo prisma/prisma.service.ts
    7.1 - implementar o OnModuleInit para iniciar a conexão quando a api iniciar
        import { Injectable, OnModuleDestroy, OnModuleInit } from '@nestjs/common';
        import { PrismaClient } from '../generated/prisma/client.js';
        import { PrismaPg } from '@prisma/adapter-pg';
        import { ConfigService } from '@nestjs/config';


        @Injectable()
        export class PrismaService extends PrismaClient implements OnModuleInit, OnModuleDestroy {

            constructor(configService: ConfigService) {
                const adapter = new PrismaPg({ connectionString: configService.get('DATABASE_URL') })
                super({
                    adapter,
                    log: ['query', 'info', 'warn', 'error'],
                });
            }

            async onModuleInit() {
                await this.$connect();
            }

            async onModuleDestroy() {
                await this.$disconnect();
            }
        }

12 - implementa os outros metodos do CRUD com o prismaService
    import { Injectable } from '@nestjs/common';
    import { CreateUserDto } from './dto/create-user.dto.js';
    import { UpdateUserDto } from './dto/update-user.dto.js';
    import { PrismaService } from '../prisma/prisma.service.js';
    import brcypt from 'bcrypt';

    @Injectable()
    export class UsersService {

        constructor(private prismaService: PrismaService) {}

    create(createUserDto: CreateUserDto) {
        return this.prismaService.user.create({
            data: {
                name: createUserDto.name,
                email: createUserDto.email,
                password: brcypt.hashSync(createUserDto.password, 10),
                role: createUserDto.role,
            },
        });
    }

    findAll() {
        return this.prismaService.user.findMany();
    }

    findOne(id: string) {
        return this.prismaService.user.findUnique({
        where: { id },
        });
    }

    update(id: string, updateUserDto: UpdateUserDto) {
        return this.prismaService.user.update({
        where: { id },
        data: updateUserDto,
        });
    }

    remove(id: string) {
        return this.prismaService.user.delete({
        where: { id },
        });
    }
    }
